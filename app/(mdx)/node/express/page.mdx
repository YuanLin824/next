## 第一阶段：认识 Express（入门）

### 1. Express 是什么？

Express 是 Node.js 最流行的 **Web 应用框架**，2010 年由 TJ Holowaychuk 创建。它封装了 Node.js 原生的 `http` 模块，提供路由、中间件、模板引擎等功能。

```
原生 http 模块                  │     Express
─────────────────────           │  ────────────────────
http.createServer((req,         │  const app = express()
  res) => {                     │  app.get('/', (req, res) => {
  if (req.url === '/'           │     res.send('Hello')
    && req.method === 'GET') {  │  })
    res.end('Hello')            │
  }                             │
})                              │
```

**核心理念：**

- **路由**：把 HTTP 方法 + URL 映射到处理器
- **中间件**：请求 → 响应之间的处理管道
- **极简**：自身不约束任何项目结构/ORM/模板

### 2. 第一个 Express 应用

```bash
npm init -y
npm install express
```

```js
const express = require("express")
const app = express()
const PORT = 3000

// 路由
app.get("/", (req, res) => {
  res.send("Hello World!")
})

app.get("/api/health", (req, res) => {
  res.json({ status: "ok", timestamp: new Date().toISOString() })
})

app.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`)
})
```

### 3. 路由基础

```js
// 路由规则：app.METHOD(PATH, HANDLER)
app.get("/users", (req, res) => {}) // 获取列表
app.post("/users", (req, res) => {}) // 创建
app.put("/users/:id", (req, res) => {}) // 完整更新
app.patch("/users/:id", (req, res) => {}) // 部分更新
app.delete("/users/:id", (req, res) => {}) // 删除

// app.all —— 匹配所有 HTTP 方法
app.all("/secret", (req, res, next) => {
  console.log("所有方法都经过这里")
  next()
})

// 通配符路由
app.get("/files/*", (req, res) => {
  res.send(`请求路径: ${req.params[0]}`) // → "docs/readme.md"
})
```

### 4. 请求对象（req）

```js
app.get("/users/:id", (req, res) => {
  // ====== URL 参数 ======
  req.params.id // → /users/42 → "42"
  req.query.page // → /users?page=2&limit=10 → "2"
  req.query.limit // → "10"

  // ====== 路径信息 ======
  req.path // → "/users/42"
  req.originalUrl // → "/users/42?page=2"（原始完整路径）
  req.baseUrl // 子路由挂载时的前缀

  // ====== 请求头 ======
  req.headers["content-type"] // → "application/json"
  req.get("content-type") // 同上（推荐）
  req.ip // 客户端 IP
  req.method // → "GET"

  // ====== 请求体（需搭配中间件） ======
  req.body // POST/PUT 请求体（需 express.json()）
})
```

### 5. 响应对象（res）

```js
app.get("/response-demo", (req, res) => {
  // ====== 发送数据 ======
  res.send("plain text") // 自动设 Content-Type: text/html
  res.json({ ok: true }) // 自动设 Content-Type: application/json
  res.sendStatus(204) // 只发状态码，不发送 body

  // ====== 设置状态码 ======
  res.status(201).json({ id: 1 })
  res.status(404).send("Not Found")

  // ====== 设置响应头 ======
  res.set("X-Custom-Header", "value")
  res.type("json") // Content-Type 快捷设置

  // ====== 重定向 ======
  res.redirect(301, "/new-url")
  res.redirect("..") // 相对重定向
  res.redirect("back") // 回到 Referer

  // ====== 文件 ======
  res.sendFile("/absolute/path/to/file.pdf")
  res.download("/path/to/file.pdf", "自定义文件名.pdf")
})
```

### 6. 获取请求体（Body Parsing）

```js
// Express 4.16+ 内置 JSON 和 URL 编码解析
app.use(express.json()) // Content-Type: application/json → req.body
app.use(express.urlencoded({ extended: true })) // application/x-www-form-urlencoded → req.body
// extended: true → 使用 qs 库（支持嵌套对象）
// extended: false → 使用 querystring 库（不支持嵌套）

// 原始 body（文本/二进制）
app.use(express.text()) // Content-Type: text/plain → req.body（字符串）
app.use(express.raw()) // Content-Type: application/octet-stream → req.body（Buffer）

// multipart/form-data（文件上传）→ express 不内置，用 multer
```

## 第二阶段：中间件（Express 的精髓）

### 7. 中间件的本质是什么？

中间件就是一个签名为 `(req, res, next)` 的函数——在**请求到达最终处理器之前**依次执行的管道。

```
请求 GET /users
    │
    ▼
┌──────────┐  next()  ┌──────────┐  next()  ┌──────────┐
│ Logger   │─────────▶│   Auth   │─────────▶│ Handler  │
│ 记录日志 │          │ 验证身份 │          │ 处理请求 │
│          │◀─────────│          │◀─────────│          │
└──────────┘          └──────────┘          └──────────┘
    │                                            │
    └────────────── 洋葱模型 ────────────────────┘

每个中间件可以做三件事：
1. 执行任何代码
2. 修改 req / res 对象
3. 调用 next() → 交给下一个中间件
   或 发回响应 → 终止后续中间件
```

```js
// 用中间件记录 IP
app.use((req, res, next) => {
  console.log(`${new Date().toISOString()} ${req.method} ${req.url} — ${req.ip}`)
  next() // ⬅ 别忘了调用 next！否则请求会挂起
})

// 用中间件拒绝非工作时间请求
app.use((req, res, next) => {
  const hour = new Date().getHours()
  if (hour < 9 || hour > 18) {
    return res.status(503).send("非工作时间")
  }
  next()
})
```

### 8. 应用级中间件 vs 路由级中间件

```js
// ====== 应用级中间件 —— 对所有路由生效 ======
app.use((req, res, next) => {
  req.requestTime = Date.now() // 给 req 附加属性，后续中间件/处理器都能读到
  next()
})

// ====== 路由级中间件 —— 只对特定路径生效 ======
app.use("/api", apiRouter) // 挂载子路由

// 直接在路由上叠加中间件
app.get(
  "/admin",
  requireAdmin, // ← 中间件 1
  loadUser, // ← 中间件 2
  (req, res) => {
    res.json(req.user) // 最终处理器
  }
)

// 路由级中间件 —— 对指定方法 + 路径生效
app.post("/users", validateUser, createUserHandler)
```

### 9. 内置中间件


### 10. 第三方中间件

```js
const morgan = require("morgan") // HTTP 访问日志
const cors = require("cors") // 跨域
const helmet = require("helmet") // 安全头
const compression = require("compression") // Gzip 压缩
const rateLimit = require("express-rate-limit") // 速率限制

// 日志——记录每个请求
app.use(morgan("dev")) // :method :url :status :response-time ms
// 生产环境：morgan('combined')

// 安全头——防御 XSS/点击劫持/嗅探
app.use(helmet())

// 压缩——减小响应体积
app.use(compression()) // 对所有响应 > 1KB 自动 gzip

// CORS —— 允许跨域
app.use(cors({ origin: "https://myfrontend.com" }))
// 或精细控制：
app.use(
  cors({
    origin: ["https://app.com", /\.trusted\.com$/],
    methods: ["GET", "POST"],
    allowedHeaders: ["Content-Type", "Authorization"],
    credentials: true, // 允许发送 Cookie
  })
)

// 速率限制——防止暴力请求
const limiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 分钟窗口
  max: 100, // 最多 100 次
  standardHeaders: true,
  legacyHeaders: false,
})
app.use("/api/", limiter) // 只对 API 路由限速
```

### 11. 错误处理中间件

```js
// 错误处理中间件的签名：4 个参数（err, req, res, next）
// Express 通过参数个数识别——必须 4 个参数！

// 自定义错误类
class AppError extends Error {
  constructor(message, statusCode) {
    super(message)
    this.statusCode = statusCode
    this.isOperational = true // 标记为可预期的业务错误
  }
}

// 全局错误处理（放最后！）
app.use((err, req, res, next) => {
  console.error(err.stack)

  const statusCode = err.statusCode || 500
  res.status(statusCode).json({
    error: {
      message: process.env.NODE_ENV === "production" ? "服务器内部错误" : err.message,
      status: statusCode,
    },
  })
})

// 404 处理（在所有路由之后）
app.use((req, res) => {
  res.status(404).json({ error: { message: "Not Found", status: 404 } })
})
```

### 12. express.Router —— 模块化路由

当路由超过 10 个时，单文件变得难以维护。用 `Router` 拆分：

```js
// routes/users.js
const router = require("express").Router()

// 所有路由都以 /users 为前缀（从 app 挂载时指定）

router.get("/", async (req, res) => {
  // GET /users
  res.json(await User.findAll())
})

router.get("/:id", async (req, res) => {
  // GET /users/42
  const user = await User.findById(req.params.id)
  if (!user) return res.status(404).json({ error: "Not Found" })
  res.json(user)
})

router.post("/", async (req, res) => {
  // POST /users
  const user = await User.create(req.body)
  res.status(201).json(user)
})

module.exports = router
```

```js
// app.js —— 主文件保持精简
```

### 13. 路由参数与高级匹配

```js
// ====== 路由参数 ======
app.get("/users/:id/posts/:postId", (req, res) => {
  // /users/42/posts/77 → { id: "42", postId: "77" }
  const { id, postId } = req.params
})

// ====== 可选参数（?） ======
app.get("/posts/:year?/:month?", (req, res) => {
  // /posts → { year: undefined, month: undefined }
  // /posts/2026 → { year: "2026" }
  // /posts/2026/07 → { year: "2026", month: "07" }
})

// ====== 正则约束参数 ======
app.get("/users/:id(\\d+)", (req, res) => {
  // 只匹配数字 ID
  // /users/42 ✅  /users/abc ❌
})

// ====== param 中间件 —— 参数预处理 ======
router.param("id", (req, res, next, id) => {
  // 任何包含 :id 参数的路由都会先经过这里
  const user = users.find((u) => u.id === parseInt(id))
  if (!user) return res.status(404).json({ error: "User not found" })
  req.user = user // 预加载到 req
  next()
})

// 之后所有含 :id 的路由都能直接用 req.user
router.get("/:id", (req, res) => res.json(req.user))
router.put("/:id", (req, res) => {
  /* req.user 已就绪 */
})
```

### 16. 文件上传（multer）

```bash
npm install multer
```

```js
const multer = require("multer")
const path = require("path")

// 配置存储
const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, "uploads/"),
  filename: (req, file, cb) => {
    const uniqueSuffix = Date.now() + "-" + Math.round(Math.random() * 1e9)
    cb(null, uniqueSuffix + path.extname(file.originalname))
  },
})

// 文件过滤
const fileFilter = (req, file, cb) => {
  if (file.mimetype.startsWith("image/")) {
    cb(null, true)
  } else {
    cb(new Error("只接受图片文件"), false)
  }
}

const upload = multer({
  storage,
  fileFilter,
  limits: { fileSize: 5 * 1024 * 1024 }, // 5MB
})

// 单文件上传
app.post("/upload", upload.single("avatar"), (req, res) => {
  // req.file → { fieldname, originalname, mimetype, size, path, filename }
  // req.body → 其他表单字段
  res.json({ url: `/uploads/${req.file.filename}` })
})

// 多文件上传
app.post("/upload/multi", upload.array("photos", 10), (req, res) => {
  res.json({ files: req.files.map((f) => f.filename) })
})
```

### 20. JWT 认证

```bash
npm install jsonwebtoken bcrypt
```

```js
const jwt = require("jsonwebtoken")
const bcrypt = require("bcrypt")
const SECRET = process.env.JWT_SECRET

// ====== 注册 ======
app.post("/register", async (req, res) => {
  const hashed = await bcrypt.hash(req.body.password, 12)
  const user = await User.create({ ...req.body, password: hashed })
  res.status(201).json({ id: user.id })
})

// ====== 登录 ======
app.post("/login", async (req, res) => {
  const user = await User.findOne({ email: req.body.email })
  if (!user || !(await bcrypt.compare(req.body.password, user.password))) {
    return res.status(401).json({ error: "邮箱或密码错误" })
  }
  const token = jwt.sign({ id: user.id, role: user.role }, SECRET, { expiresIn: "7d" })
  res.json({ token })
})

// ====== 认证中间件 ======
function authenticate(req, res, next) {
  const auth = req.headers.authorization
  if (!auth?.startsWith("Bearer ")) {
    return res.status(401).json({ error: "未提供 Token" })
  }
  try {
    const payload = jwt.verify(auth.split(" ")[1], SECRET)
    req.user = payload
    next()
  } catch {
    res.status(401).json({ error: "Token 无效或已过期" })
  }
}

// ====== 授权中间件 ======
function authorize(...roles) {
  return (req, res, next) => {
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: "权限不足" })
    }
    next()
  }
}

// 使用
app.get("/admin/users", authenticate, authorize("admin"), async (req, res) => {
  res.json(await User.findAll())
})
```

```js
// 用高阶函数包装 async handler，消除重复 try-catch
function asyncHandler(fn) {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next)

}

// 或直接用 express-async-errors（最简单方案）
require("express-async-errors")
```

### 24. 安全加固清单

```js
const helmet = require("helmet")
const rateLimit = require("express-rate-limit")
const cors = require("cors")
const hpp = require("hpp") // 防 HTTP 参数污染

// 1. 安全头
app.use(helmet())

// 2. CORS 白名单
app.use(cors({ origin: process.env.CORS_ORIGIN, credentials: true }))

// 3. 速率限制
app.use("/api/auth", rateLimit({ windowMs: 15 * 60 * 1000, max: 20 }))

// 4. 防参数污染（?a=1&a=2 → 只保留最后一个）
app.use(hpp())

// 5. 隐藏 Express 签名
app.disable("x-powered-by")

// 6. 限制请求体大小
app.use(express.json({ limit: "100kb" }))

// 7. 信任代理（部署在 Nginx/ALB 后正确获取 IP）
app.set("trust proxy", 1)
```

### 25. 优雅关机

```js
// server.js（分离 app 和 server）
const app = require("./app")

const server = app.listen(process.env.PORT || 3000)

// 捕获未处理的 Promise rejection
process.on("unhandledRejection", (reason) => {
  console.error("UNHANDLED REJECTION:", reason)
})

// 优雅关机
process.on("SIGTERM", () => {
  console.log("SIGTERM received — 开始优雅关机")
  server.close(() => {
    console.log("HTTP server closed")
    // 关闭数据库连接
    // mongoose.connection.close()
    // redis.quit()
    process.exit(0)
  })

  // 30 秒后强制退出
  setTimeout(() => {
    console.error("强制退出")
    process.exit(1)
  }, 30000)
})
```

### 28. 常见面试问题

| 问题                       | 要点                                                          |
| -------------------------- | ------------------------------------------------------------- |
| Express 中间件的本质？     | `(req, res, next)` 函数链，洋葱模型                           |
| `app.use` vs `app.get`？   | `app.use` 匹配路径前缀（所有方法），`app.get` 只匹配 GET      |
| 错误处理中间件怎么写？     | 4 参数 `(err, req, res, next)`，放在所有路由最后              |
| next() 不调用会怎样？      | 请求挂起，超时                                                |
| 如何模块化路由？           | `express.Router()` + `app.use("/prefix", router)`             |
| body-parser 需要单独装吗？ | Express 4.16+ 内置 `express.json()` 和 `express.urlencoded()` |
