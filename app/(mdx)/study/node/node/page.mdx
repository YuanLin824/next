# Node.js 学习指南

> 难度递进式 Node.js 学习文档，从环境搭建到生产部署。

## 第一阶段：基础入门

### 1. 什么是 Node.js？

Node.js 是 **V8 引擎 + C++ 绑定的 libuv 事件循环**构成的 JavaScript 运行时。让 JS 脱离浏览器运行在服务端。

```
JavaScript 代码 → V8 引擎 (执行 JS) → libuv (异步I/O/事件循环) → 操作系统
```

### 2. 安装与运行

```bash
node -v           # 查看版本
node app.js       # 运行文件
node -e "console.log(1+2)"  # 执行片段
```

```js
// app.js — 第一个 Node.js 程序
console.log("Hello Node.js")
console.log(__dirname) // 当前文件所在目录（绝对路径）
console.log(__filename) // 当前文件绝对路径
console.log(process.cwd()) // 进程工作目录
```

### 3. 模块系统 — CommonJS

```js
// math.js — 导出
exports.add = (a, b) => a + b
module.exports = { add, multiply: (a, b) => a * b }

// app.js — 导入
const math = require("./math") // 文件模块
const fs = require("fs") // 内置模块
const express = require("express") // npm 模块

// 模块缓存: 同一个模块只加载一次，第二次 require 返回缓存
// 查找顺序: 内置模块 → node_modules → 文件(.js/.json/.node)
```

### 4. ES Module（Node.js 12+）

```js
// package.json: { "type": "module" }
// 或文件名 .mjs

import fs from "fs"
import { add } from "./math.js"

export const VERSION = "1.0"
export default class User {}
```

> **小结**：能运行 Node.js 代码，理解 CommonJS 和 ESM 两种模块系统。

## 第二阶段：内置模块（常用）

### 5. path — 路径处理

```js
const path = require("path")

path.join("/a", "/b", "c") // "/a/b/c" — 拼接路径（自动处理分隔符）
path.resolve("a", "b") // "/current/dir/a/b" — 解析为绝对路径
path.dirname("/a/b/c.js") // "/a/b" — 目录名
path.basename("/a/b/c.js") // "c.js" — 文件名
path.basename("/a/b/c.js", ".js") // "c" — 去除扩展名
path.extname("file.txt") // ".txt" — 扩展名
path.parse("/a/b/c.js") // { root, dir, base, ext, name }
```

### 6. fs — 文件系统

```js
const fs = require("fs")
const fsp = require("fs/promises")

// 同步（阻塞，仅初始化时用）
const data = fs.readFileSync("./file.txt", "utf-8")

// 回调
fs.readFile("./file.txt", "utf-8", (err, data) => {
  if (err) {
    console.error(err)
    return
  }
  console.log(data)
})

// Promise（推荐）
const data = await fsp.readFile("./file.txt", "utf-8")
await fsp.writeFile("./output.txt", "Hello World")

// 流式读写（大文件）
const rs = fs.createReadStream("./large.txt")
const ws = fs.createWriteStream("./copy.txt")
rs.pipe(ws)

// 目录操作
await fsp.mkdir("./new-dir", { recursive: true }) // 递归创建
const files = await fsp.readdir("./dir") // 读取目录
await fsp.rm("./dir", { recursive: true }) // 删除目录

// 文件状态
const stat = await fsp.stat("./file.txt")
stat.isFile() // 是否文件
stat.isDirectory() // 是否目录
stat.size // 文件大小
```

### 7. os — 系统信息

```js
const os = require("os")

os.cpus() // CPU 信息
os.totalmem() // 总内存
os.freemem() // 可用内存
os.homedir() // 用户目录
os.platform() // 平台 (darwin/linux/win32)
os.tmpdir() // 临时目录
os.userInfo() // 当前用户信息
```

### 8. process — 进程

```js
process.cwd() // 工作目录
process.argv // 命令行参数 [node, script, ...args]
process.env // 环境变量
process.env.NODE_ENV // 常用: development / production
process.pid // 进程 ID
process.uptime() // 运行秒数
process.memoryUsage() // { heapTotal, heapUsed, rss }
process.exit(0) // 退出 (0=成功, 1=失败)

// 事件
process.on("exit", (code) => console.log(`退出: ${code}`))
process.on("uncaughtException", (err) => console.error("未捕获:", err))
process.on("unhandledRejection", (reason) => console.error("未处理 Promise:", reason))
```

> **小结**：掌握 path/fs/os/process 四个核心内置模块。

## 第三阶段：HTTP 与网络

### 9. http — 创建 HTTP 服务

```js
const http = require("http")

const server = http.createServer((req, res) => {
  const { method, url, headers } = req

  // 设置响应头
  res.writeHead(200, { "Content-Type": "application/json" })

  // 解析请求体
  let body = ""
  req.on("data", (chunk) => (body += chunk))
  req.on("end", () => {
    console.log(JSON.parse(body))
    res.end(JSON.stringify({ status: "ok" }))
  })
})

server.listen(3000, () => console.log("http://localhost:3000"))
```

### 10. url — URL 解析

```js
const url = require("url")

const parsed = new URL("https://example.com:8080/path?q=hello#hash")
parsed.hostname // "example.com"
parsed.pathname // "/path"
parsed.searchParams.get("q") // "hello"
```

### 11. 构建 RESTful API（原生 http）

```js
const server = http.createServer((req, res) => {
  const { method, url } = req
  const parsed = new URL(url, `http://${req.headers.host}`)

  if (method === "GET" && parsed.pathname === "/api/users") {
    res.writeHead(200, { "Content-Type": "application/json" })
    res.end(JSON.stringify({ users: [{ id: 1, name: "Alice" }] }))
    return
  }

  if (method === "POST" && parsed.pathname === "/api/users") {
    let body = ""
    req.on("data", (c) => (body += c))
    req.on("end", () => {
      res.writeHead(201)
      res.end(JSON.stringify({ created: JSON.parse(body) }))
    })
    return
  }

  res.writeHead(404).end("Not Found")
})
```

> **小结**：理解 HTTP 服务本质（请求-响应模型），学会 URL 解析和请求体处理。

```bash

```

```js
const express = require("express")
const app = express()

app.use(express.json()) // 解析 JSON 请求体
app.use(express.static("public")) // 静态文件

// 路由
app.get("/api/users", (req, res) => res.json({ users: [] }))
app.post("/api/users", (req, res) => {
  const user = req.body
  res.status(201).json({ created: user })
})
app.get("/api/users/:id", (req, res) => {
  const { id } = req.params // 路由参数
  const { page } = req.query // 查询参数 ?page=1
  res.json({ id, page })
})

app.listen(3000)
```

### 16. Express 中间件

```js
// 应用级中间件
app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`)
  next() // 必须调用 next 才能继续
})

// 路由级中间件
const auth = (req, res, next) => {
  if (!req.headers.authorization) return res.status(401).json({ error: "未授权" })
  next()
}
app.get("/api/admin", auth, (req, res) => res.json({}))

// 错误处理中间件（4 参数）
app.use((err, req, res, next) => {
  console.error(err.stack)
  res.status(500).json({ error: err.message })
})
```

### 17. 路由模块化

```js

router.get("/", (req, res) => res.json({ users: [] }))
router.post("/", (req, res) => res.status(201).json({}))
router.put("/:id", (req, res) => res.json({ id: req.params.id }))
router.delete("/:id", (req, res) => res.status(204).send())

// app.js

```

## 第七阶段：流 (Stream)

### 21. 四种基本流

```js
const { Readable, Writable, Transform, Duplex } = require("stream")

// Readable — 可读流
const rs = fs.createReadStream("./input.txt")
rs.on("data", (chunk) => console.log(chunk.length))
rs.on("end", () => console.log("读取完毕"))

// Writable — 可写流
const ws = fs.createWriteStream("./output.txt")
ws.write("Hello\n")
ws.end("World")

// Transform — 转换流
const { Transform } = require("stream")
const upper = new Transform({
  transform(chunk, encoding, callback) {
    callback(null, chunk.toString().toUpperCase())
  },
})

// pipe 链: 读 → 转换 → 写
fs.createReadStream("./input.txt").pipe(upper).pipe(fs.createWriteStream("./output.txt"))

// pipeline — 自动处理错误和清理
const { pipeline } = require("stream/promises")
await pipeline(fs.createReadStream("./input.txt"), upper, fs.createWriteStream("./output.txt"))
```

### 22. 自定义流

```js
// 自定义可读流
class Counter extends Readable {
  constructor(max) {
    super()
    this.max = max
    this.i = 0
  }
  _read() {
    this.i++
    if (this.i > this.max)
      this.push(null) // 结束
    else this.push(String(this.i))
  }
}

// 自定义可写流
class Logger extends Writable {
  _write(chunk, encoding, callback) {
    console.log(chunk.toString().trim())
    callback()
  }
}

new Counter(5).pipe(new Logger()) // 1 2 3 4 5
```

> **小结**：理解 Readable/Writable/Transform/Duplex 四种流，掌握 pipe 和 pipeline。

## 第八阶段：集群与多进程

### 23. child_process — 子进程

```js
const { exec, execFile, spawn, fork } = require("child_process")

// exec — 执行 shell 命令（缓冲输出，适合短命令）
exec("ls -la", (err, stdout, stderr) => console.log(stdout))

// spawn — 流式输出（适合长时间运行、大量输出）
const child = spawn("ls", ["-la"])
child.stdout.on("data", (data) => console.log(`stdout: ${data}`))

// fork — 衍生新的 Node.js 进程（自动 IPC 通信）
const worker = fork("./worker.js")
worker.send({ task: "heavyComputation", data: [1, 2, 3] })
worker.on("message", (result) => console.log(result))
```

### 24. cluster — 多核利用

```js
const cluster = require("cluster")
const http = require("http")
const os = require("os")

if (cluster.isPrimary) {
  console.log(`主进程 ${process.pid} 运行`)
  os.cpus().forEach(() => cluster.fork()) // 每 CPU 一个 Worker

  cluster.on("exit", (worker) => {
    console.log(`Worker ${worker.process.pid} 退出，重新启动`)
    cluster.fork()
  })
} else {
  // Worker 进程 — 共享同一个端口
  http.createServer((req, res) => res.end(`Worker ${process.pid}`)).listen(3000)
}
```

### 25. worker_threads — 线程（共享内存）

```js
const { Worker, isMainThread, parentPort, workerData } = require("worker_threads")

// 主线程
if (isMainThread) {
  const worker = new Worker(__filename, { workerData: { start: 0, end: 1000000 } })
  worker.on("message", (result) => console.log("结果:", result))
} else {
  // 工作线程
  let sum = 0
  for (let i = workerData.start; i < workerData.end; i++) sum += i
  parentPort.postMessage(sum)
}
```

> **小结**：child_process 子进程、cluster 多进程、worker_threads 多线程的区别和适用场景。

## 第九阶段：事件循环与性能

### 26. libuv 事件循环 6 个阶段

```
   ┌───────────────────────────┐
┌─>│        timers             │  setTimeout / setInterval
│  ├───────────────────────────┤
│  │   pending callbacks       │  延迟的 I/O 回调
│  ├───────────────────────────┤
│  │     idle, prepare         │  内部
│  ├───────────────────────────┤
│  │         poll              │  获取新的 I/O 事件
│  ├───────────────────────────┤
│  │         check             │  setImmediate
│  ├───────────────────────────┤
│  │    close callbacks        │  socket.on('close')
└──┴───────────────────────────┘
        ┌───────────────┐
        │ nextTick 队列 │  优先级最高
        │ 微任务队列    │  Promise.then
        └───────────────┘
```

```js
// nextTick > Promise.then > setTimeout/setImmediate
console.log("1")
process.nextTick(() => console.log("2"))
Promise.resolve().then(() => console.log("3"))
setTimeout(() => console.log("4"), 0)
setImmediate(() => console.log("5"))
// 输出: 1 2 3 5 4（或 1 2 3 4 5）
// 说明: nextTick 优先于 Promise，setImmediate 和 setTimeout(0) 顺序取决于执行阶段
```

### 27. 性能优化要点

```bash
# 1. 避免同步 API（fs.readFileSync 等阻塞整个进程）
# 2. 合理设置 cluster Worker 数量 (= CPU 核心数)
# 3. 用 stream 处理大文件（避免内存爆炸）
# 4. 数据库连接池复用（避免频繁创建/销毁）
# 5. 查询结果加 limit、建索引
# 6. 用 Promise.all 并行不依赖的异步操作
# 7. 避免在循环中使用 await（用 Promise.all + map）
# 8. 用 process.memoryUsage() 和 clinic.js 排查内存泄漏
# 9. 开启 HTTP keep-alive（复用 TCP 连接）
# 10. 生产环境 NODE_ENV=production
```

> **小结**：深刻理解事件循环 6 阶段和 nextTick"插队"机制，掌握 10 条性能优化。

### process 常用属性和事件

| ---------------------------------- | ------------------- |
| `process.argv`                     | 命令行参数          |

### cluster vs worker_threads vs child_process

| 维度     | cluster       | worker_threads       | child_process |
| -------- | ------------- | -------------------- | ------------- |
| 隔离级别 | 进程          | 线程                 | 进程          |
| 内存共享 | ❌            | ✅ SharedArrayBuffer | ❌            |
| 通信方式 | IPC           | postMessage          | IPC/stdio     |
| 适合     | HTTP 服务扩展 | CPU 密集型计算       | 调用外部程序  |

### 文件系统 API 同步 vs 异步

| 操作     | 同步            | 回调        | Promise         |
| -------- | --------------- | ----------- | --------------- |
| 读文件   | `readFileSync`  | `readFile`  | `fsp.readFile`  |
| 写文件   | `writeFileSync` | `writeFile` | `fsp.writeFile` |
| 创建目录 | `mkdirSync`     | `mkdir`     | `fsp.mkdir`     |
| 读取目录 | `readdirSync`   | `readdir`   | `fsp.readdir`   |

### 36. 异步模式进阶

```js
// 1. 并发限制器
async function mapLimit(items, limit, fn) {
  const results = []
  const executing = new Set()
  for (const item of items) {
    const p = Promise.resolve()
      .then(() => fn(item))
      .then((r) => {
        results.push(r)
        executing.delete(p)
      })
    executing.add(p)
    if (executing.size >= limit) await Promise.race(executing)
  }
  await Promise.all(executing)
  return results
}

// 2. 异步任务队列
class AsyncQueue {
  constructor() {
    this.tasks = []
    this.running = false
  }
  add(task) {
    this.tasks.push(task)
    this.run()
  }
  async run() {
    if (this.running) return
    this.running = true
    while (this.tasks.length) {
      await this.tasks.shift()()
    }
    this.running = false
  }
}

// 3. 带超时的 Promise
function withTimeout(promise, ms) {
  return Promise.race([
    promise,
    new Promise((_, reject) => setTimeout(() => reject(new Error("超时")), ms)),
  ])
}
```

## 第二十阶段：缓存策略

### 52. 多级缓存实现

```js
const cache = new Map()

// 内存缓存 + Redis
async function getCached(key, ttlSeconds, fetchFn) {
  // L1: 内存
  if (cache.has(key)) {
    const { value, expiry } = cache.get(key)
    if (Date.now() < expiry) return value
    cache.delete(key)
  }

  // L2: Redis
  const redisValue = await redis.get(key)
  if (redisValue) {
    const parsed = JSON.parse(redisValue)
    cache.set(key, { value: parsed, expiry: Date.now() + ttlSeconds * 1000 })
    return parsed
  }

  // L3: 源数据
  const data = await fetchFn()
  cache.set(key, { value: data, expiry: Date.now() + ttlSeconds * 1000 })
  await redis.set(key, JSON.stringify(data), "EX", ttlSeconds)
  return data
}

// 使用
const users = await getCached("users:list", 60, () => db.query("SELECT * FROM users"))
```

### 61. 零停机重启

```js
// cluster 模式零停机重启
if (cluster.isPrimary) {
  for (let i = 0; i < os.cpus().length; i++) cluster.fork()

  process.on("SIGHUP", () => {
    console.log("收到 SIGHUP，滚动重启 Worker")
    const workers = Object.values(cluster.workers)
    let i = 0

    function restartNext() {
      const worker = workers[i++]
      if (!worker) return console.log("全部重启完成")
      const newWorker = cluster.fork()
      newWorker.on("listening", () => {
        worker.disconnect() // 旧 Worker 不再接收新连接
        setTimeout(() => worker.kill(), 5000) // 5 秒后杀掉（等已有请求处理完）
        restartNext()
      })
    }
    restartNext()
  })
}
```
