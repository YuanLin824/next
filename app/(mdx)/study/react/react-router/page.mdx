## 第一阶段：安装与项目结构（入门）

### 1. React Router v7 框架模式

结合了 Remix 的全栈能力，提供**文件路由**、**SSR/SSG/SPA** 渲染策略、**loader/action** 数据加载与变更。

```
环境: Node.js ≥ 22.22.0 + React ≥ 19.0.0
```

### 2. 创建项目

```bash
npx create-react-router@latest my-app
cd my-app && npm install && npm run dev
```

### 3. 核心文件

```
app/
├── routes.ts          # 路由配置（URL ↔ 模块映射，核心文件）
├── root.tsx           # 根布局（HTML 骨架 + 全局错误边界）
├── routes/            # 路由模块目录
│   └── home.tsx
├── entry.client.tsx   # 客户端入口（hydrateRoot）
├── entry.server.tsx   # 服务端入口（renderToPipeableStream）
├── react-router.config.ts  # 框架配置（SSR/SPA/预渲染）
└── vite.config.ts          # Vite 配置
```

### 4. 路由模块基础（route module）

每个路由文件可导出：`default`（组件，必需）、`loader`（数据加载）、`action`（数据变更）、`meta`（SEO）、`ErrorBoundary`（错误边界）。

```tsx
// app/routes/home.tsx
import type { Route } from "./+types/home"

export async function loader({ request }: Route.LoaderArgs) {
  return { message: "Hello" }
}

export default function Home({ loaderData }: Route.ComponentProps) {
  return <h1>{loaderData.message}</h1>
}
```

> **小结**：理解框架模式的项目结构，知道 `routes.ts`、`root.tsx`、路由模块三个核心概念。

## 第二阶段：路由配置（核心能力）

### 5. routes.ts — 集中式路由配置

```ts
import { type RouteConfig, route, index, layout, prefix } from "@react-router/dev/routes"

export default [
  index("./routes/home.tsx"), // 首页 /
  route("about", "./routes/about.tsx"), // /about
  route("users/:userId", "./routes/user-detail.tsx"), // 动态参数
  route("files/*", "./routes/files.tsx"), // 通配符
  route("search/:query?", "./routes/search.tsx"), // 可选参数

  layout("./routes/auth/layout.tsx", [
    // 布局路由
    route("login", "./routes/auth/login.tsx"),
  ]),

  ...prefix("dashboard", [
    // 路径前缀
    index("./routes/dashboard/home.tsx"),
    route("settings", "./routes/dashboard/settings.tsx"),
  ]),

  route("*", "./routes/not-found.tsx"), // 404
] satisfies RouteConfig
```

### 6. 路由辅助函数

| 函数                     | 用途                      |
| ------------------------ | ------------------------- |
| `route(pattern, file)`   | 标准路由                  |
| `index(file)`            | 父路径精确匹配            |
| `layout(file, children)` | 布局路由（不增加 URL 段） |
| `prefix(path, routes)`   | 共享前缀，无父布局文件    |
| `relative(dirname)`      | 在子目录中创建相对路由    |

### 7. 文件路由（约定式，可选）

```ts
// 不想写 routes.ts 时可用文件约定
import { flatRoutes } from "@react-router/fs-routes"
export default flatRoutes() satisfies RouteConfig
```

```
app/routes/
├── _index.tsx           → /
├── about.tsx            → /about
├── users.$userId.tsx    → /users/:userId
├── dashboard._index.tsx → /dashboard
└── $.tsx                → /* (404)
```

> **小结**：能用 route/index/layout/prefix 配置路由，理解动态段 `:id` 和通配符 `*`。

## 第三阶段：数据加载（loader / clientLoader）

### 8. loader — 服务端数据获取

```tsx
export async function loader({ params, request }: Route.LoaderArgs) {
  const url = new URL(request.url)
  const page = url.searchParams.get("page") ?? "1"

  const res = await fetch(`https://api.example.com/products?page=${page}`)
  if (!res.ok) throw new Response("加载失败", { status: res.status })
  return { products: await res.json(), page }
}

export default function Products({ loaderData }: Route.ComponentProps) {
  return (
    <ul>
      {loaderData.products.map((p) => (
        <li key={p.id}>{p.name}</li>
      ))}
    </ul>
  )
}
```

**核心规则**：loader 在组件渲染**之前**执行，组件拿到数据时已经就绪——**消除加载瀑布流**。

### 9. clientLoader — 客户端数据获取

```tsx
// 场景 1: 纯客户端数据获取
export async function clientLoader({ params }: Route.ClientLoaderArgs) {
  return fetch(`/api/products/${params.pid}`).then((r) => r.json())
}
export function HydrateFallback() {
  return <div>加载中...</div>
}

// 场景 2: loader + clientLoader 组合
export async function loader({ params }: Route.LoaderArgs) {
  return fetchProduct(params.pid!)
}
export async function clientLoader({ serverLoader }: Route.ClientLoaderArgs) {
  const cached = sessionStorage.getItem(`product-${params.pid}`)
  if (cached) return JSON.parse(cached)
  const data = await serverLoader()
  sessionStorage.setItem(`product-${params.pid}`, JSON.stringify(data))
  return data
}
clientLoader.hydrate = true // 首次水合也执行
```

### 10. clientLoader.hydrate 模式

| 配置                                | 行为                                        |
| ----------------------------------- | ------------------------------------------- |
| 仅 `loader`                         | 首屏 SSR，后续导航自动 fetch loader         |
| `clientLoader`（`hydrate = false`） | 首屏用 SSR 数据，后续导航用 clientLoader    |
| `clientLoader.hydrate = true`       | 首屏也用 clientLoader，显示 HydrateFallback |
| 仅 `clientLoader`                   | 纯客户端路由                                |

### 11. shouldRevalidate — 控制重新验证

```tsx
export function shouldRevalidate({ currentParams, nextParams }: Route.ShouldRevalidateArgs) {
  return currentParams.id !== nextParams.id // 仅 id 变化时重新加载
}
```

> **小结**：掌握 loader/clientLoader 两种数据获取方式，理解 hydrate 三种模式。

## 第四阶段：数据变更（action + Form）

### 12. action — 处理表单提交

```tsx
import { Form, redirect, useActionData } from "react-router"

export async function action({ request }: Route.ActionArgs) {
  const formData = await request.formData()
  const email = formData.get("email") as string
  const password = formData.get("password") as string

  const errors: Record<string, string> = {}
  if (!email) errors.email = "请输入邮箱"
  if (!password) errors.password = "请输入密码"
  if (Object.keys(errors).length > 0) return { errors } // 返回数据，不进 ErrorBoundary

  const res = await fetch("/api/login", {
    method: "POST",
    body: JSON.stringify({ email, password }),
  })
  if (!res.ok) return { errors: { server: "邮箱或密码错误" } }

  return redirect("/dashboard") // 成功 → 重定向
}

export default function Login({ actionData }: Route.ComponentProps) {
  return (
    <Form method="post">
      <input name="email" /> {actionData?.errors?.email && <p>{actionData.errors.email}</p>}
      <input name="password" type="password" />
      {actionData?.errors?.server && <p>{actionData.errors.server}</p>}
      <button type="submit">登录</button>
    </Form>
  )
}
```

### 13. `<Form>` 组件的行为

- `method="post"` → 自动调用 action（不需 `onSubmit`、不需手动 `fetch`）
- `method="get"` → 等同于 `<Link>`，触发搜索
- action 完成后**所有活跃路由的 loader 自动重新验证**

### 14. useFetcher — 不导航的数据交互

```tsx
// 收藏按钮（触发 action，不改变 URL）
function FavoriteButton({ productId }: { productId: string }) {
  const fetcher = useFetcher()
  return (
    <fetcher.Form method="post" action={`/products/${productId}/favorite`}>
      <button>{fetcher.state === "idle" ? "❤️" : "⏳"}</button>
    </fetcher.Form>
  )
}

// 搜索建议（调用 loader，不改变 URL）
function SearchSuggestions({ query }: { query: string }) {
  const fetcher = useFetcher<{ suggestions: string[] }>()
  useEffect(() => {
    if (query.length >= 2) fetcher.load(`/search-suggestions?q=${query}`)
  }, [query])
  return fetcher.state === "loading" ? (
    <Spinner />
  ) : (
    <ul>
      {fetcher.data?.suggestions.map((s) => (
        <li key={s}>{s}</li>
      ))}
    </ul>
  )
}
```

### 15. 乐观更新（Optimistic UI）

```tsx
function MessageList({ loaderData }: Route.ComponentProps) {
  const fetcher = useFetcher()
  const optimisticMsg = fetcher.formData?.get("message")
  const messages = optimisticMsg
    ? [...loaderData.messages, { id: "temp", text: String(optimisticMsg) }]
    : loaderData.messages

  return (
    <>
      <ul>
        {messages.map((m) => (
          <li key={m.id} style={{ opacity: m.id === "temp" ? 0.5 : 1 }}>
            {m.text}
          </li>
        ))}
      </ul>
      <fetcher.Form method="post">
        <input name="message" />
        <button>发送</button>
      </fetcher.Form>
    </>
  )
}
```

> **小结**：action+Form 处理表单，useFetcher 处理不导航的变更，理解乐观更新。

## 第六阶段：错误处理

### 18. 错误冒泡机制

```
root.tsx ErrorBoundary ← 最终兜底
├── dashboard/layout ErrorBoundary ← 可在此层捕获
│   ├── analytics.tsx ← 错误发生在此
│   │   ErrorBoundary ← 先在此捕获（最近）
│   └── settings.tsx ← 不受影响（兄弟路由）
```

- 错误在**最近的路由 ErrorBoundary** 中被捕获
- 无 ErrorBoundary → 往上冒泡到父路由
- 只有**错误路由子树**被替换，兄弟路由不受影响

### 19. ErrorBoundary 实现

```tsx
export function ErrorBoundary({ error }: Route.ErrorBoundaryProps) {
  if (isRouteErrorResponse(error)) {
    return (
      <div>
        <h1>
          {error.status} — {error.statusText}
        </h1>
        <p>{error.data}</p>
      </div>
    )
  }
  if (error instanceof Error) {
    return (
      <div>
        <h1>出错了</h1>
        <p>{error.message}</p>
      </div>
    )
  }
  return <h1>未知错误</h1>
}
```

## 第八阶段：渲染策略 (SSR/SPA/SSG)

### 22. react-router.config.ts 配置

```ts
import type { Config } from "@react-router/dev/config"

// SSR（默认）
export default { ssr: true } satisfies Config

// SPA
export default { ssr: false } satisfies Config

// SSG 预渲染
export default {
  ssr: true,
  async prerender() {
    return ["/", "/about", "/blog/hello-world"]
  },
} satisfies Config

// 混合模式
export default {
  ssr: false,
  async prerender() {
    return ["/"]
  }, // 首页静态，其余 SPA
} satisfies Config
```

| 策略     | `ssr`   | `prerender` | 适用              |
| -------- | ------- | ----------- | ----------------- |
| **SSR**  | `true`  | 可选        | 全栈应用、SEO     |
| **SPA**  | `false` | 无          | 后台管理、仪表盘  |
| **SSG**  | `true`  | URL 列表    | 博客、文档        |
| **混合** | `false` | 部分路径    | 首页静态+应用 SPA |

### 30. entry.server.tsx（流式 SSR）

```tsx
import { ServerRouter } from "react-router"
import { renderToPipeableStream } from "react-dom/server"
import { PassThrough } from "node:stream"
import { isbot } from "isbot"

export default function handleRequest(request, status, headers, routerContext) {
  return isbot(request.headers.get("user-agent"))
    ? handleBotRequest(request, status, headers, routerContext) // 爬虫：完整 HTML
    : handleBrowserRequest(request, status, headers, routerContext) // 浏览器：流式传输
}

function handleBrowserRequest(request, status, headers, routerContext) {
  return new Promise((resolve) => {
    const { pipe, abort } = renderToPipeableStream(
      <ServerRouter context={routerContext} url={request.url} />,
      {
        onShellReady() {
          const body = new PassThrough()
          headers.set("Content-Type", "text/html")
          resolve(new Response(body, { status, headers }))
          pipe(body)
        },
        onError(err) {
          console.error(err)
        },
      }
    )
    request.signal.addEventListener("abort", abort)
  })
}
```

| 概念           | 说明                             |
| -------------- | -------------------------------- |
| `onShellReady` | 壳 HTML 准备好即开始流式传输     |
| `onAllReady`   | 完整 HTML 后一次性返回（爬虫用） |
| `PassThrough`  | Node.js 流管道                   |
| `abort`        | 客户端断开 → 停止渲染            |

```tsx
function InfiniteList() {
  const fetcher = useFetcher<{ items: Item[]; nextCursor: string }>()
  const [items, setItems] = useState<Item[]>([])

  useEffect(() => {
    if (fetcher.data) setItems((prev) => [...prev, ...fetcher.data!.items])
  }, [fetcher.data])

  const loadMore = () => fetcher.load(`/items?cursor=${fetcher.data?.nextCursor ?? ""}`)

  return (
    <>
      {items.map((item) => (
        <ItemCard key={item.id} item={item} />
      ))}
      <button onClick={loadMore} disabled={fetcher.state !== "idle"}>
        {fetcher.state === "loading" ? "加载中..." : "加载更多"}
      </button>
    </>
  )
}
```

### 38. 搜索防抖（useFetcher + useDebounce）

```tsx
function SearchPage() {
  const fetcher = useFetcher<{ results: { id: string; name: string }[] }>()
  const [query, setQuery] = useState("")
  const debounced = useDebounce(query, 300)

  useEffect(() => {
    if (debounced.length >= 2) fetcher.load(`/search?q=${encodeURIComponent(debounced)}`)
  }, [debounced])

  return (
    <>
      <input value={query} onChange={(e) => setQuery(e.target.value)} />
      {fetcher.state === "loading" && <Spinner />}
      {fetcher.data?.results.map((r) => (
        <div key={r.id}>{r.name}</div>
      ))}
    </>
  )
}
```

### 39. 文件上传（useFetcher + FormData）

```tsx
function FileUpload() {
  const fetcher = useFetcher<{ url: string }>()
  const [preview, setPreview] = useState<string>()

  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return
    setPreview(URL.createObjectURL(file))
    const formData = new FormData()
    formData.append("file", file)
    fetcher.submit(formData, { method: "POST", action: "/upload", encType: "multipart/form-data" })
  }

  return (
    <div>
      <input type="file" onChange={handleFile} />
      {fetcher.state === "submitting" && <Spinner />}
      {preview && <img src={preview} />}
      {fetcher.data?.url && <p>上传成功: {fetcher.data.url}</p>}
    </div>
  )
}
```

### 40. Cookie 操作（在 loader/action 中）

```tsx
// 读 Cookie
export async function loader({ request }: Route.LoaderArgs) {
  const cookieHeader = request.headers.get("Cookie")
  const locale = parseCookie(cookieHeader)?.locale ?? "zh"
  return { locale }
}

// 设 Cookie（重定向时）
export async function action({ request }: Route.ActionArgs) {
  const formData = await request.formData()
  const locale = formData.get("locale") as string
  const cookie = serialize("locale", locale, { path: "/", maxAge: 365 * 24 * 60 * 60 })
  return redirect("/", { headers: { "Set-Cookie": cookie } })
}
```

### 41. 结合 Zod 数据校验

```tsx
import { z } from "zod"

const SignupSchema = z.object({
  email: z.string().email("邮箱格式错误"),
  password: z.string().min(8, "密码至少 8 位"),
})

export async function action({ request }: Route.ActionArgs) {
  const formData = Object.fromEntries(await request.formData())
  const result = SignupSchema.safeParse(formData)
  if (!result.success) {
    return { errors: result.error.flatten().fieldErrors }
  }
  await db.user.create({ data: result.data })
  return redirect("/welcome")
}
```
